\textbf{\texttt{Register Direct}}\par
\vspace{2pt}
\begin{manualeaprofile}
\manualeaprofilesyntax{Rn(r)}
\manualeaprofilesyntax{SP}
\manualeaprofileline{EA encoding}{000rrrr selects Rn(r); 1101000 selects SP.}
\manualeaprofileline{EA type}{register operand, not a memory reference.}
\manualeaprofileline{Generation}{The operand value is the selected register value. No effective memory address is generated.}
\manualeaprofileline{Segment}{No segment is selected.}
\manualeaprofileline{Payload}{No appended EA payload bytes.}
\manualeaprofileline{Update}{No auto-update.}
\end{manualeaprofile}
\manualeadirectflow{Direct register operand}{REGISTER}{Rn or SP contents}{REGISTER VALUE}
\end{manualformblock}
\begin{manualformblock}{5.07in}
\textbf{\texttt{Rn Memory}}\par
\vspace{2pt}
\begin{manualeaprofile}
\manualeaprofilesyntax{[Rn(r)]}
\manualeaprofilesyntax{[Rn(r) + disp8s]}
\manualeaprofilesyntax{[Rn(r) + disp16s]}
\manualeaprofilesyntax{[Rn(r) + disp32s]}
\manualeaprofilesyntax{[Rn(r) + disp64]}
\manualeaprofileline{EA encoding}{001rrrr has no displacement; 010rrrr, 011rrrr, 100rrrr, and 101rrrr select disp8s, disp16s, disp32s, and disp64.}
\manualeaprofileline{EA type}{memory operand.}
\manualeaprofileline{Generation}{offset = temporary Rn(r) + displacement. The no-displacement form uses displacement 0.}
\manualeaprofileline{Segment}{Uses the operation default data segment unless the instruction defines a different default.}
\manualeaprofileline{Payload}{Only displacement forms append payload bytes. Displacement payload is little-endian and has the size named by the EA form.}
\manualeaprofileline{Update}{No auto-update in compact Rn memory forms.}
\end{manualeaprofile}
\manualeaadditivememoryflow{Rn memory address generation}{BASE REGISTER}{Rn(r)}{optional displacement}{EFFECTIVE ADDRESS}
\end{manualformblock}

\begin{manualformblock}{2.35in}
\textbf{\texttt{EXT0 Escape}}\par
\vspace{2pt}
\begin{manualeaprofile}
\manualeaprofilesyntax{EXT0}
\manualeaprofilesyntax{EXT0/disp8s}
\manualeaprofilesyntax{EXT0/disp16s}
\manualeaprofilesyntax{EXT0/disp32s}
\manualeaprofilesyntax{EXT0/disp64}
\manualeaprofileline{EA encoding}{1110100 selects no displacement; 1110000, 1110001, 1110010, and 1110011 select disp8s, disp16s, disp32s, and disp64.}
\manualeaprofileline{EA type}{memory operand described by the appended EXT0 descriptor.}
\manualeaprofileline{Generation}{The EXT0 descriptor selects segment, base, index, and auto-update behavior. The compact EA escape selects only the displacement size.}
\manualeaprofileline{Segment}{EXT0 has no implicit segment. Segment-qualified forms encode SREG SEG(s); SP and PC indexed forms use SS and CS; default base-update forms use the operation default data segment.}
\manualeaprofileline{Payload}{Payload order is EXT0 descriptor byte or bytes first, then the optional displacement payload selected by the compact EA escape.}
\manualeaprofileline{Update}{Only EXT0 forms with ++ or {-}{-} update the temporary operand-evaluation image.}
\end{manualeaprofile}
\end{manualformblock}

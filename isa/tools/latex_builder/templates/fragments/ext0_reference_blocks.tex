\textbf{\texttt{EXT0 Explicit Segment Base}}\par
\vspace{2pt}
\begin{manualeaprofile}
\manualeaprofilesyntax{[SEG(s):Rn(b) + displacement]}
\manualeaprofileline{Descriptor}{0sssbbbb, one byte.}
\manualeaprofileline{Generation}{offset = temporary Rn(b) + displacement.}
\manualeaprofileline{Segment}{SEG(s) selects DS, SS, or GS0..GS5; CS is not an explicit selector.}
\manualeaprofileline{Payload}{One descriptor byte plus the optional displacement selected by the compact EXT0 escape.}
\manualeaprofileline{Update}{No auto-update.}
\end{manualeaprofile}
\manualeaadditivememoryflow{EXT0 explicit segment base}{BASE REGISTER}{Rn(b)}{optional displacement}{EFFECTIVE ADDRESS}

\begin{manualformblock}{6.43in}
\textbf{\texttt{EXT0 Explicit Segment Indexed}}\par
\vspace{2pt}
\begin{manualeaprofile}
\manualeaprofilesyntax{[SEG(s):Rn(b) + Rn(i) * scale + displacement]}
\manualeaprofilesyntax{[SEG(s):Rn(b) + Rn(i)++ * scale + displacement]}
\manualeaprofilesyntax{[SEG(s):Rn(b) + {-}{-}Rn(i) * scale + displacement]}
\manualeaprofileline{Descriptor}{Byte 0 is \texttt{1sss0010}, \texttt{1sss0000}, or \texttt{1sss0001} for no update, postincrement, or predecrement. Byte 1 is \texttt{bbbbiiii}.}
\manualeaprofileline{Generation}{offset = temporary Rn(b) + index\_term * scale + displacement.}
\manualeaprofileline{Segment}{SEG(s) selects DS, SS, or GS0..GS5; CS is not an explicit selector.}
\manualeaprofileline{Scale}{Scale is implicit in the consuming instruction. B/W/L/Q normally imply 1/2/4/8.}
\manualeaprofileline{Payload}{Two descriptor bytes plus the optional displacement selected by the compact EXT0 escape.}
\manualeaprofileline{Update}{Index postincrement uses the current temporary Rn(i), then increments it by one element. Index predecrement decrements temporary Rn(i) by one element before use.}
\end{manualeaprofile}
\manualeaindexedmemoryflow{EXT0 explicit segment indexed}{BASE REGISTER}{Rn(b)}{optional displacement}{Rn(i)}{EFFECTIVE ADDRESS}
\end{manualformblock}

\begin{manualformblock}{6.43in}
\textbf{\texttt{EXT0 SP/PC Indexed}}\par
\vspace{2pt}
\begin{manualeaprofile}
\manualeaprofilesyntax{[SP + Rn(i) * scale + displacement]}
\manualeaprofilesyntax{[SP + Rn(i)++ * scale + displacement]}
\manualeaprofilesyntax{[SP + {-}{-}Rn(i) * scale + displacement]}
\manualeaprofilesyntax{[PC + Rn(i) * scale + displacement]}
\manualeaprofilesyntax{[PC + Rn(i)++ * scale + displacement]}
\manualeaprofilesyntax{[PC + {-}{-}Rn(i) * scale + displacement]}
\manualeaprofileline{Descriptor}{Byte 0 is \texttt{10001010} for SP or \texttt{10001011} for PC. Byte 1 is \texttt{0010iiii}, \texttt{0000iiii}, or \texttt{0001iiii} for no update, postincrement, or predecrement.}
\manualeaprofileline{Generation}{offset = temporary SP or instruction-start PC + index\_term * scale + displacement. A PC term names byte 0 of the current instruction.}
\manualeaprofileline{Segment}{SP forms are fixed to SS. PC forms are fixed to CS.}
\manualeaprofileline{Payload}{Two descriptor bytes plus the optional displacement selected by the compact EXT0 escape.}
\manualeaprofileline{Update}{Only the Rn(i) index register may auto-update; SP and PC are not auto-updated by these forms.}
\end{manualeaprofile}
\manualeaindexedmemoryflow{EXT0 SP/PC indexed}{SP OR PC}{SP or PC term}{optional displacement}{Rn(i)}{EFFECTIVE ADDRESS}
\end{manualformblock}
